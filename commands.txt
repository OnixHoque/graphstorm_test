prepare_fb15k.py

import json
from pathlib import Path

import pandas as pd
from torchkge.utils.datasets import load_fb15k

# Load FB15k training dataset
print("Loading FB15k...", flush=True)
kg, _, _ = load_fb15k()

print(f"Entities: {kg.n_ent}")
print(f"Relations: {kg.n_rel}")
print(f"Training triples: {kg.n_facts}")

# Output directory
out = Path("fb15k").resolve()
out.mkdir(parents=True, exist_ok=True)

# Save entity IDs
pd.DataFrame({
    "id": range(kg.n_ent)
}).to_csv(out / "entities.csv", index=False)

# Save edges, one edge type per relation
edges = []

for r in range(kg.n_rel):
    mask = kg.relations == r

    df = pd.DataFrame({
        "src": kg.head_idx[mask].cpu().numpy(),
        "dst": kg.tail_idx[mask].cpu().numpy(),
    })

    path = out / f"relation_{r}.csv"
    df.to_csv(path, index=False)

    edges.append({
        "source_id_col": "src",
        "dest_id_col": "dst",
        "relation": ["entity", f"rel_{r}", "entity"],
        "files": str(path),
        "format": {
            "name": "csv"
        },
        "labels": [{
            "task_type": "link_prediction",
            "split_pct": [1.0, 0.0, 0.0]
        }]
    })

# GraphStorm construction config
config = {
    "version": "gconstruct-v0.1",
    "nodes": [{
        "node_type": "entity",
        "node_id_col": "id",
        "files": str(out / "entities.csv"),
        "format": {
            "name": "csv"
        }
    }],
    "edges": edges
}

# Write graph configuration
config_path = out / "graph.json"

with open(config_path, "w") as f:
    json.dump(config, f, indent=2)

print("\nPreparation complete!", flush=True)
print(f"Graph config: {config_path}")
print(f"Entities: {kg.n_ent}")
print(f"Relations: {kg.n_rel}")
print(f"Edge types: {len(edges)}")
